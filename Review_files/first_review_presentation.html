AI-Assisted 3D Assembly Design

Predicting Missing & Next Components in CAD Assemblies using Graph Neural Networks

Abstract — Engineering CAD assemblies are inherently graph-structured: components form nodes and assembly constraints form edges. This project builds an intelligent recommendation system that trains a Graph Attention Network (GAT) on historical assembly graphs sourced from the open ABC Dataset and PartNet. The model addresses two tasks: (1) detecting missing components via link prediction on partial graphs, and (2) suggesting the most suitable next component via top-K embedding-space ranking. A LinkPredictor MLP head decodes pairwise node embeddings into edge-existence probabilities, while a NodeRanker uses cosine similarity over candidate embeddings to produce a ranked shortlist — all within a pure-Python, PyTorch Geometric pipeline with no CAD API dependency. Interpretability is provided through GNNExplainer subgraph visualisations.

Data Pipeline & Graph Schema

① Raw Data Ingestion

STEP files parsed via pythonOCC / metadata JSON from PartNet. Extract component bounding boxes, volume, surface area, and mate constraint lists.

② Node Feature Engineering

Each node encodes: component type one-hot (6 classes), normalised volume, surface area, bbox dimensions (x, y, z) → 11-dim feature vector per node.

③ Edge Feature Engineering

Each mate constraint → bidirectional edge with 2-dim attribute: [mate-type encoded, weight=1.0]. Types: coincident, concentric, parallel, tangent, fixed.

④ Train / Val / Test Splits

70 / 15 / 15 split by assembly ID. Partial graphs generated by randomly masking 20–40 % of edges (link prediction) or terminal nodes (ranking task).

Graph schema for one assembly

Node dim: 11  |  Edge dim: 2  |  Avg nodes/graph: ~18  |  Avg edges: ~32

Graph Attention Network & Link Prediction

Graph Attention Network (GAT) — Veličković et al., 2018

Each node attends to its neighbours with learned weights. For node i attending to neighbour j:

/* attention coefficient */
eij = LeakyReLU( aT [Whi ‖ Whj] )
αij = softmaxj(eij)
/* new node embedding */
h'i = σ( Σj∈N(i) αij · Whj )
4 Attention Heads hidden = 128 out = 64 ELU activation

Multi-Head Attention Benefits

  • Captures multiple relationship types simultaneously
  • Attention weights provide built-in interpretability
  • Robust to noisy or irrelevant neighbours

Link Prediction — Graph Autoencoder Approach

Encoder produces node embeddings Z. A decoder scores candidate edges:

/* GAE dot-product decoder */
Â = σ( Z · ZT )
/* MLP decoder (this project) */
score(u,v) = MLP( hu ‖ hv )
/* Binary cross-entropy loss */
ℒ = BCE( score, y ) + λ·||W||²

import torch from torch_geometric.data import Data, Dataset COMP_TYPES = ['body', 'fastener', 'bearing', 'shaft', 'plate', 'housing'] MATE_TYPES = ['coincident', 'concentric', 'parallel', 'tangent', 'fixed'] def build_assembly_graph(components, mates): """ Build a PyG Data object from raw assembly metadata. components : list of dicts {id, type, volume, surface_area, bbox:[x,y,z]} mates : list of tuples (src_id, dst_id, mate_type) """ # ── Node feature matrix ────────────────────────────────────────── id_map = {c['id']: i for i, c in enumerate(components)} node_feats = [] for comp in components: type_oh = [1 if comp['type'] == t else 0 for t in COMP_TYPES] geom_vec = [comp['volume'], comp['surface_area'], comp['bbox'][0], comp['bbox'][1], comp['bbox'][2]] node_feats.append(geom_vec + type_oh) # 11-dim per node # ── Edge index + edge attributes (bidirectional) ───────────────── src_list, dst_list, edge_attr = [], [], [] for src_id, dst_id, mate_type in mates: s, d = id_map[src_id], id_map[dst_id] m_enc = MATE_TYPES.index(mate_type) if mate_type in MATE_TYPES else len(MATE_TYPES) for u, v in [(s, d), (d, s)]: # undirected → two directed edges src_list.append(u); dst_list.append(v) edge_attr.append([m_enc, 1.0]) return Data( x = torch.tensor(node_feats, dtype=torch.float), edge_index = torch.tensor([src_list, dst_list], dtype=torch.long), edge_attr = torch.tensor(edge_attr, dtype=torch.float) ) class ABCAssemblyDataset(Dataset): def __init__(self, root, transform=None): super().__init__(root, transform) def len(self): return len(self.processed_file_names) def get(self, idx): return torch.load(self.processed_dir + f'/data_{idx}.pt')

GNN Model Definition  model.py

model.py
import torch, torch.nn as nn, torch.nn.functional as F from torch_geometric.nn import GATConv, BatchNorm class AssemblyGNN(nn.Module): """Shared GAT encoder — produces 64-dim node embeddings.""" def __init__(self, in_dim, hidden=128, out_dim=64, heads=4, dropout=0.2): super().__init__() self.dropout = dropout self.conv1 = GATConv(in_dim, hidden, heads=heads, dropout=dropout) self.bn1 = BatchNorm(hidden * heads) self.conv2 = GATConv(hidden * heads, out_dim, heads=1, concat=False, dropout=dropout) def forward(self, x, edge_index): x = F.elu(self.bn1(self.conv1(x, edge_index))) # (N, 512) x = F.dropout(x, p=self.dropout, training=self.training) return self.conv2(x, edge_index) # (N, 64) class LinkPredictor(nn.Module): """MLP head: score a candidate edge (u, v) → missing component detection.""" def __init__(self, embed_dim=64): super().__init__() self.net = nn.Sequential( nn.Linear(embed_dim * 2, 128), nn.ReLU(), nn.Dropout(0.1), nn.Linear(128, 1) ) def forward(self, z, edge_index): h = torch.cat([z[edge_index[0]], z[edge_index[1]]], dim=-1) return self.net(h).squeeze(-1) # (E,) raw logits class NodeRanker(nn.Module): """Rank candidate nodes by cosine similarity to the partial-graph context.""" def forward(self, z_partial, z_candidates): # mean-pool partial assembly to get context vector ctx = z_partial.mean(dim=0, keepdim=True) # (1, 64) scores = F.cosine_similarity( ctx.expand(z_candidates.size(0), -1), z_candidates ) return scores # (C,) per candidate def build_model(in_dim): gnn = AssemblyGNN(in_dim) predictor = LinkPredictor() ranker = NodeRanker() return gnn, predictor, ranker

from torch_geometric.utils import negative_sampling from sklearn.metrics import roc_auc_score, average_precision_score import torch.nn.functional as F def train_one_epoch(gnn, predictor, loader, optimiser, device): gnn.train(); predictor.train() total_loss = 0.0 for batch in loader: batch = batch.to(device) optimiser.zero_grad() z = gnn(batch.x, batch.edge_index) # encode all nodes pos_score = predictor(z, batch.edge_index) # positive edges neg_ei = negative_sampling( # hard negatives batch.edge_index, num_nodes = batch.num_nodes, num_neg_samples = batch.edge_index.size(1) ) neg_score = predictor(z, neg_ei) labels = torch.cat([ torch.ones(pos_score.size(0)), torch.zeros(neg_score.size(0)) ]).to(device) loss = F.binary_cross_entropy_with_logits( torch.cat([pos_score, neg_score]), labels ) loss.backward(); optimiser.step() total_loss += loss.item() return total_loss / len(loader) @torch.no_grad() def evaluate(gnn, predictor, data, device): gnn.eval(); predictor.eval() data = data.to(device) z = gnn(data.x, data.edge_index) pos_score = torch.sigmoid(predictor(z, data.edge_index)).cpu() neg_ei = negative_sampling(data.edge_index, num_nodes=data.num_nodes) neg_score = torch.sigmoid(predictor(z, neg_ei)).cpu() scores = torch.cat([pos_score, neg_score]).numpy() labels = [1]*len(pos_score) + [0]*len(neg_score) return { 'auc': roc_auc_score(labels, scores), 'ap' : average_precision_score(labels, scores) } if __name__ == '__main__': device = 'cuda' if torch.cuda.is_available() else 'cpu' gnn, predictor, ranker = build_model(in_dim=11) gnn.to(device); predictor.to(device) optimiser = torch.optim.Adam( list(gnn.parameters()) + list(predictor.parameters()), lr=1e-3, weight_decay=5e-4 ) for epoch in range(1, 201): loss = train_one_epoch(gnn, predictor, train_loader, optimiser, device) metrics = evaluate(gnn, predictor, val_data, device) print(f"Epoch {epoch:03d} | loss={loss:.4f} | AUC={metrics['auc']:.4f}")
